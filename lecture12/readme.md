Аллокаторы  - это шаблонный класс который маскирует вызовы new и delete 

Стандарный аллокатор (4 основных метода)

template <class T>
struct std::allocator<T> {
    T* allocate(size_t n) const {
        return (T*) (new char[n*sizeof(T)]);
    }
    void deallocate(T* p, size_t n) const {
        delete[] char* p;
    }
    template<typename ... Args> 
    void construct (T* p, const Args& ... args) const {
        return new(p) T(args...);
    }
    void destoy(T* p) const {
        p-> ~T();
    }
}
