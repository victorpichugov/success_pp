10) CRTP (это паттерн)
Curious Recurring Template Pattern

пусть у меня есть класс Base

template<class T> // шаблонный класс
struct Base {
    void f() { // 1 версия
        static_cast<T*>(this) -> f();}
};

struct Derived: public Base<Derived> { // это бред(( но надо разобраться
    void f() {} // 2 версия
}


Derived d;
Base* b = d;
b.f(); // какую f вызовет b? вторую. Это имитация виртуальных функций без слова virtual
